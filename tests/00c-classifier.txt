const candidate = $('Prepare Website Probe').first().json;
const response = $json || {};
const status = Number(response.statusCode || response.status || response.error?.statusCode || response.error?.httpCode || 0);
const err = response.error;
const cause = err && typeof err === 'object' ? err.cause : null;
const networkCode = String(cause?.code || err?.code || response.code || '').toUpperCase();
const errorDetail = [typeof err === 'string' ? err : err?.message,
  cause?.message, networkCode, response.errorMessage, response.message]
  .filter(Boolean).join(' | ').slice(0, 1200);
const fetchErrorCode = /ENOTFOUND|EAI_AGAIN/.test(networkCode + ' ' + errorDetail)
  ? 'WEBSITE_DNS_FAILED' : /TIMEOUT|TIMEDOUT|ETIMEDOUT/i.test(networkCode + ' ' + errorDetail)
  ? 'WEBSITE_CONNECTION_TIMEOUT' : /ECONNREFUSED/.test(networkCode + ' ' + errorDetail)
  ? 'WEBSITE_CONNECTION_REFUSED' : 'WEBSITE_FETCH_FAILED';
const bodyRaw = response.body ?? response.data ?? '';
const html = typeof bodyRaw === 'string' ? bodyRaw.slice(0, 2000000) : JSON.stringify(bodyRaw || {}).slice(0, 2000000);
const clean = (value, max = 3000) => String(value ?? '').replace(/\s+/g, ' ').trim().slice(0, max);
const decode = (value) => clean(String(value || '').replace(/&amp;/gi, '&').replace(/&quot;/gi, '"').replace(/&#39;/gi, "'").replace(/&lt;/gi, '<').replace(/&gt;/gi, '>'), 3000);
const fold = (value) => clean(value, 500000).toLowerCase().normalize('NFKD').replace(/[^\w\s]/g, ' ').replace(/_/g, ' ').replace(/\s+/g, ' ').trim();
const extract = (regex) => decode((html.match(regex) || [,''])[1] || '');
const title = extract(/<title[^>]*>([\s\S]*?)<\/title>/i);
const metaDescription = extract(/<meta[^>]+(?:name|property)\s*=\s*["'](?:description|og:description)["'][^>]+content\s*=\s*["']([^"']*)["'][^>]*>/i) || extract(/<meta[^>]+content\s*=\s*["']([^"']*)["'][^>]+(?:name|property)\s*=\s*["'](?:description|og:description)["'][^>]*>/i);
const h1 = extract(/<h1[^>]*>([\s\S]*?)<\/h1>/i).replace(/<[^>]+>/g, ' ');
const hasViewport = /<meta[^>]+name\s*=\s*["']viewport["'][^>]*>/i.test(html);
const hasContactCta = /\b(contact us|call us|book (?:an )?appointment|schedule (?:an )?appointment|request (?:an )?appointment|get in touch|free consultation)\b/i.test(html);
const text = fold(html.replace(/<script[\s\S]*?<\/script>/gi, ' ').replace(/<style[\s\S]*?<\/style>/gi, ' ').replace(/<[^>]+>/g, ' '));
const titleFold = fold(title);
const stop = new Set(['the','and','of','at','in','on','for','a','an','llc','inc','ltd','pllc','pc','company','co']);
const nameTokens = [...new Set(fold(candidate.company_name).split(' ').filter((token) => token.length >= 3 && !stop.has(token)))];
const titleMatches = nameTokens.filter((token) => titleFold.includes(token));
const bodyMatches = nameTokens.filter((token) => text.includes(token));
const titleCoverage = nameTokens.length ? titleMatches.length / nameTokens.length : 0;
const bodyCoverage = nameTokens.length ? bodyMatches.length / nameTokens.length : 0;
const phoneDigits = clean(candidate.phone, 80).replace(/\D/g, '');
const phoneTail = phoneDigits.slice(-7);
const phoneMatch = Boolean(phoneTail && html.replace(/\D/g, '').includes(phoneTail));
const cityMatch = Boolean(candidate.city && text.includes(fold(candidate.city)));
const streetNumber = (clean(candidate.address, 300).match(/\b\d{2,6}\b/) || [])[0] || '';
const streetMatch = Boolean(streetNumber && text.includes(streetNumber));
const domainMatch = clean(candidate.probe_url, 2000).match(/^https?:\/\/(?:[^@\/\s]+@)?([^\/?#:\s]+)(?::\d+)?(?:[\/?#]|$)/i);
const finalProbeUrl = clean(response.finalUrl || response.url || candidate.probe_url,2000);
const finalDomainMatch = finalProbeUrl.match(/^https?:\/\/([^\/?#:]+)(?:[\/?#]|$)/i);
const domain = String(finalDomainMatch?.[1] || domainMatch?.[1] || '').toLowerCase().replace(/^www\./, '').replace(/\.$/, '');
const emailDomain = clean(candidate.email, 320).toLowerCase().split('@').pop();
const emailDomainMatch = Boolean(emailDomain && domain === emailDomain);
const blocked = [...['att.net','gmail.com','outlook.com','hotmail.com','live.com','yahoo.com','yahoo.net','currently.att.yahoo.com'],"google.com","googleusercontent.com","g.page","yelp.com","facebook.com","instagram.com","linkedin.com","youtube.com","x.com","twitter.com","tiktok.com","yellowpages.com","mapquest.com","healthgrades.com","zocdoc.com","vitals.com","doximity.com","bbb.org","chamberofcommerce.com","opencorporates.com","crunchbase.com","tripadvisor.com","angi.com","thumbtack.com","nextdoor.com","wikipedia.org","waze.com","snapfinance.com","tintingnearme.com","miamigoeslocal.com","nears.me","manta.com","bizapedia.com","cylex.us.com","alignable.com","merchantcircle.com","local.yahoo.com","foursquare.com","superpages.com","hotfrog.com","brownbook.net","showmelocal.com","whereorg.com",'dentalplans.com','wheree.com','zoominfo.com','webmd.com','doctor.com','carecredit.com','birdeye.com','trustpilot.com','npidb.org','npino.com','npi-lookup.org','npi-lookup.com','npiprofile.com','nppes.cms.hhs.gov','npi.wiki','doximity.com','medicare.gov','yandex.com','point2homes.com','denteldoc.com','wellness.com','dentists.com'];
const domainBlocked = blocked.some((entry) => domain === entry || domain.endsWith('.' + entry));
const domainLabel = fold(domain.split('.')[0] || '');
const brandMatches = nameTokens.filter((token) => domainLabel.includes(token));
const brandCoverage = nameTokens.length ? brandMatches.length / nameTokens.length : 0;
const urlText = clean(candidate.probe_url, 2000).toLowerCase();
const directoryLike = domainBlocked ||
    /\b(?:npi|registry|registries|provider database|license lookup|employee directory)\b/i.test(title) ||
    /\/(?:organizations|directory|profiles?|registry|npi|maps\/org|government|courts?)(?:\/|[.?_-])/i.test(urlText) || /(?:\/|\b)(?:browse|directory|directories|listing|listings|search|find-a|find-an|find-stores|providers?|live-map)(?:[\/.?_-]|\b)/i.test(urlText);
const genericBrand = new Set(['dental','dentist','dentists','clinic','care','health','medical','services','service','plumbing','plumber','hotel','hotels','restaurant','restaurants','salon','spa','business','group','center','centre']);
const cityTokens = new Set(fold(candidate.city).split(' '));
const distinctiveTokens = nameTokens.filter(token => !genericBrand.has(token) && !cityTokens.has(token));
const distinctiveDomainMatch = distinctiveTokens.some(token => domainLabel.includes(token));
const fullTitleMatch = nameTokens.length > 0 && nameTokens.every(token => titleFold.split(' ').includes(token));
const distinctiveBodyMatch = distinctiveTokens.some(token => text.split(' ').includes(token));
const brandedHomepage = (distinctiveDomainMatch && distinctiveBodyMatch && (phoneMatch || cityMatch || titleCoverage >= 0.5)) ||
  (fullTitleMatch && brandCoverage >= 0.34 && (phoneMatch || cityMatch));
const branchBrandMatch = distinctiveDomainMatch && distinctiveBodyMatch && titleMatches.length >= 2 && titleCoverage >= 0.5 && (phoneMatch || cityMatch && titleCoverage >= 0.75);
const contactBackedBrandMatch = phoneMatch && titleCoverage >= 0.5 && bodyCoverage >= 0.5 && (distinctiveDomainMatch || brandCoverage >= 0.5);
const strongIdentity = !directoryLike && Boolean(brandedHomepage || branchBrandMatch || contactBackedBrandMatch);
let confidence = 0; // Fresh page evidence, not the previous search estimate.
confidence += Math.round(titleCoverage * 30) + Math.round(bodyCoverage * 20);
if (phoneMatch) confidence += 25;
if (cityMatch) confidence += 8;
if (streetMatch) confidence += 7;
// Matching an email provider domain is not identity evidence.
if (brandCoverage >= 0.34) confidence += 15;
if (directoryLike) confidence -= 60;
if(strongIdentity) confidence = Math.max(confidence, phoneMatch ? 95 : 85);
confidence = Math.max(0, Math.min(100, confidence));
const retryable = status === 0 || [408,425,429].includes(status) || status >= 500;
const challenge = /^(just a moment|access denied|attention required|request blocked|verify you are human|robot check)/i.test(title.trim());
const accessible = status >= 200 && status < 300 && html.length >= 50 && !challenge;
const restricted = [401,403].includes(status);
const ssl = finalProbeUrl.startsWith('https://');
const yearMatches = [...html.matchAll(/(?:©|&copy;|copyright)\s*(?:\d{4}\s*[-–]\s*)?(\d{4})/gi)].map((match) => Number(match[1])).filter((year) => year >= 1990 && year <= new Date().getUTCFullYear() + 1);
const copyrightYear = yearMatches.length ? Math.max(...yearMatches) : null;
if (retryable) return [{ json: {
  ...candidate, retryable_error: true,
  probe_error_code: status === 429 ? 'WEBSITE_RATE_LIMITED' :
    (status >= 500 ? 'WEBSITE_SERVER_ERROR' : fetchErrorCode),
  probe_error: ('Homepage probe failed: ' + (status ? 'HTTP ' + status :
    (errorDetail || 'No HTTP response received'))).slice(0,1500),
  website_http_status_next: status || null,
  website_accessible_next: status ? false : null,
  identity_evidence_next: { ...(candidate.website_identity_evidence || {}),
    last_probe: { url: candidate.probe_url, http_status: status || null,
      network_code: networkCode || null, error: errorDetail || null,
      checked_at: new Date().toISOString() } }
} }];
let verificationStatus = 'needs_review';
let condition = restricted ? 'access_restricted' : (!accessible ? 'website_unreachable' : (directoryLike ? 'directory_or_aggregator' : 'identity_uncertain'));
let needSignal = restricted ? 'Website blocks automated verification; manual review required' : (!accessible ? 'Website URL is not currently accessible; manual review required' : (directoryLike ? 'Directory or aggregator page is not accepted as the official business website' : 'Website identity match is uncertain; manual review required'));
if (accessible && !directoryLike && confidence >= 70 && strongIdentity) {
  verificationStatus = 'verified';
  const issues = [];
  if (!title) issues.push('missing page title');
  if (!metaDescription) issues.push('missing meta description');
  if (!hasViewport) issues.push('missing mobile viewport');
  if (!h1) issues.push('missing H1');
  if (!hasContactCta) issues.push('no clear contact or booking CTA');
  condition = issues.length ? 'basic_issues_detected' : 'verified_basic';
  needSignal = issues.length ? 'Website verified; ' + issues.join(', ') : 'Website verified; deep performance audit pending';
}
return [{ json: {
  ...candidate,
  retryable_error: false,
  verification_status_next: verificationStatus,
  website_condition_next: condition,
  need_signal_next: needSignal.slice(0, 1000),
  website_to_save: verificationStatus === 'verified' ? finalProbeUrl : (candidate.website || null),
  domain_to_save: verificationStatus === 'verified' ? domain : (candidate.domain || null),
  resolved_website_next: finalProbeUrl,
  resolved_domain_next: domain,
  identity_confidence_next: accessible ? confidence : Number(candidate.website_identity_confidence || 0),
  verification_error_code_next: restricted ? 'WEBSITE_ACCESS_RESTRICTED' : (!accessible ? 'WEBSITE_HTTP_ERROR' : null),
  verification_error_next: restricted ? 'Homepage returned HTTP ' + status + '; automated access denied. Manual review required.' : (!accessible ? 'Homepage returned HTTP ' + status + '; manual review required.' : null),
  identity_evidence_next: { ...(candidate.website_identity_evidence || {}), homepage: { http_status: status, accessible, title_token_coverage: Number(titleCoverage.toFixed(2)), body_token_coverage: Number(bodyCoverage.toFixed(2)), phone_match: phoneMatch, city_match: cityMatch, street_number_match: streetMatch, email_domain_match: emailDomainMatch, blocked_domain: domainBlocked, directory_like: directoryLike, brand_domain_coverage: Number(brandCoverage.toFixed(2)), strong_identity: strongIdentity } },
  website_http_status_next: status || null,
  website_accessible_next: accessible,
  website_has_ssl_next: ssl,
  website_title_next: accessible ? title : null,
  website_meta_description_next: accessible ? metaDescription : null,
  website_h1_next: accessible ? h1 : null,
  website_has_viewport_next: accessible ? hasViewport : null,
  website_has_contact_cta_next: accessible ? hasContactCta : null,
  website_copyright_year_next: accessible ? copyrightYear : null
} }];
